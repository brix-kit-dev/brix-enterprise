## 2. 租户配置能力设计（TenantConfigCapability）

> **核心问题**：租户需要自主配置默认语言、主题、时区、通知偏好等信息，这些配置影响所有租户成员的默认体验。

### 2.2 租户配置项定义

```typescript
/**
 * 租户配置 — 基础功能范围内的完整配置项
 * 
 * 参考: AWS Organizations Settings, Azure AD Tenant Properties, Salesforce Org Defaults
 */

// ============================
// 租户基础设置（sys_tenant 表扩展字段）
// ============================
interface TenantSettings {
  // --- 区域与本地化 ---
  defaultLocale: string;            // 默认语言/地区: "zh-CN" | "ja-JP" | "en-US"
  defaultTimezone: string;          // 默认时区: "Asia/Shanghai" | "Asia/Tokyo" | "UTC"
  defaultDateFormat: string;        // 日期格式: "YYYY-MM-DD" | "MM/DD/YYYY" | "DD/MM/YYYY"
  defaultTimeFormat: '12h' | '24h'; // 时间格式
  defaultCurrency: string;          // 默认货币: "CNY" | "JPY" | "USD"

  // --- 外观与品牌 ---
  defaultTheme: 'light' | 'dark' | 'system'; // 默认主题
  brandingConfig: TenantBrandingConfig;

  // --- 安全策略 ---
  sessionTimeoutMinutes: number;    // 会话超时（分钟），默认 60
  passwordPolicy: PasswordPolicy;
  mfaPolicy: 'disabled' | 'optional' | 'required';  // MFA 策略
  allowedLoginMethods: LoginMethod[]; // 允许的登录方式

  // --- 通知偏好 ---
  notificationChannels: NotificationChannel[]; // 启用的通知渠道
  businessHours: BusinessHours;      // 工作时间（影响通知静默期）
}

interface TenantBrandingConfig {
  logoUrl?: string;                  // 租户 Logo URL
  faviconUrl?: string;               // Favicon URL
  primaryColor: string;              // 主色调, 默认 "#1976d2"
  secondaryColor?: string;           // 次要色调
  loginPageTitle?: string;           // 登录页标题
  loginPageSubtitle?: string;        // 登录页副标题
  loginPageBgUrl?: string;           // 登录页背景图
}

interface PasswordPolicy {
  minLength: number;                 // 最小长度，默认 8
  requireUppercase: boolean;         // 要求大写字母
  requireLowercase: boolean;         // 要求小写字母
  requireNumbers: boolean;           // 要求数字
  requireSpecialChars: boolean;      // 要求特殊字符
  maxAgeDays: number;                // 密码过期天数，0=不过期
  historyCount: number;              // 不能重复最近 N 个密码
}

type LoginMethod = 'phone_sms' | 'email_password' | 'username_password' | 'wechat' | 'dingtalk' | 'google' | 'github';
type NotificationChannel = 'in_app' | 'email' | 'sms' | 'push';

interface BusinessHours {
  timezone: string;                  // 工作时间所在时区
  weekdays: number[];                // 工作日 [1,2,3,4,5] = 周一到周五
  startTime: string;                 // 开始时间 "09:00"
  endTime: string;                   // 结束时间 "18:00"
}

// ============================
// 用户个人偏好（biz_user_profile.preferences）
// ============================
interface UserPreferences {
  locale?: string;                   // 覆盖租户默认语言
  timezone?: string;                 // 覆盖租户默认时区
  dateFormat?: string;               // 覆盖租户默认日期格式
  timeFormat?: '12h' | '24h';       // 覆盖租户默认时间格式
  theme?: 'light' | 'dark' | 'system'; // 覆盖租户默认主题
  notificationPreferences?: {
    channels: NotificationChannel[];
    quietHoursEnabled: boolean;
    quietHoursStart?: string;        // "22:00"
    quietHoursEnd?: string;          // "08:00"
  };
}
```

### 2.3 数据库 Schema 变更

```sql
-- ============================================
-- 扩展 sys_tenant 表 — 新增配置字段
-- ============================================
-- Flyway: V010__add_tenant_settings.sql
-- （V007-V009 已被 B2B2C 模型占用，详见第 13 章）

ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_locale      VARCHAR(20)  DEFAULT 'zh-CN';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_timezone    VARCHAR(50)  DEFAULT 'Asia/Shanghai';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_date_format VARCHAR(20)  DEFAULT 'YYYY-MM-DD';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_time_format VARCHAR(5)   DEFAULT '24h';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_currency    VARCHAR(10)  DEFAULT 'CNY';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS default_theme       VARCHAR(10)  DEFAULT 'light';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS session_timeout_min INT          DEFAULT 60;
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS mfa_policy          VARCHAR(20)  DEFAULT 'optional';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS allowed_login_methods JSONB      DEFAULT '["phone_sms", "email_password"]';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS password_policy     JSONB        DEFAULT '{"minLength":8,"requireUppercase":false,"requireLowercase":true,"requireNumbers":true,"requireSpecialChars":false,"maxAgeDays":0,"historyCount":0}';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS notification_channels JSONB      DEFAULT '["in_app"]';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS business_hours      JSONB        DEFAULT '{}';
ALTER TABLE sys_tenant ADD COLUMN IF NOT EXISTS settings            JSONB        DEFAULT '{}'; -- 扩展配置兜底

-- ============================================
-- sys_tenant_config 表 — 插件级租户配置
-- ============================================
-- Flyway: V011__create_sys_tenant_config.sql

CREATE TABLE IF NOT EXISTS sys_tenant_config (
    id              BIGINT PRIMARY KEY,
    tenant_id       BIGINT NOT NULL,
    
    config_namespace VARCHAR(100) NOT NULL,        -- 配置命名空间（插件名）: "platform" | "reservation" | "medical"
    config_key      VARCHAR(200) NOT NULL,         -- 配置键: "appointment.default_duration_min"
    config_value    JSONB NOT NULL,                -- 配置值（JSON 支持复杂结构）
    config_type     VARCHAR(20) NOT NULL DEFAULT 'STRING',  -- STRING | NUMBER | BOOLEAN | JSON | ENUM
    
    description     VARCHAR(500),                  -- 配置描述
    is_sensitive    BOOLEAN DEFAULT FALSE,          -- 是否敏感（加密存储 / 脱敏展示）
    is_readonly     BOOLEAN DEFAULT FALSE,          -- 是否只读（仅平台管理员可改）
    
    created_at      TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_by      BIGINT,
    
    CONSTRAINT uk_tenant_config UNIQUE (tenant_id, config_namespace, config_key),
    CONSTRAINT fk_tenant_config_tenant FOREIGN KEY (tenant_id) REFERENCES sys_tenant(id)
);

CREATE INDEX idx_tenant_config_tenant ON sys_tenant_config(tenant_id);
CREATE INDEX idx_tenant_config_ns ON sys_tenant_config(tenant_id, config_namespace);
```

## 3. 跨端多租户鉴权设计

> **核心问题**：同一个人用同一手机号在平台注册了两个租户的成员，在移动端和Web端如何鉴权？如何切换？

### 3.1 核心原则

```text
┌─────────────────────────────────────────────────────────────────────────────────────────┐
│               跨端多租户鉴权核心原则（参考 Auth0 Organizations + Slack 模型）             │
├─────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                          │
│  原则1: 身份全局唯一，成员关系多对多                                                     │
│  ════════════════════════════════════════════════════════════════════════════════════   │
│  • 1 个手机号 = 1 个 Identity（全局唯一身份）                                            │
│  • 1 个 Identity 可以属于 N 个 Tenant（通过 sys_tenant_member）                         │
│  • 1 个 Identity 在不同 Tenant 可以有不同角色（patient vs staff）                       │
│  • Token 内必须包含当前 Tenant ID，不允许"无租户 Token" 访问业务 API                    │
│                                                                                          │
│  原则2: 认证与授权两阶段分离                                                             │
│  ════════════════════════════════════════════════════════════════════════════════════   │
│  • 阶段1 — 认证（Who are you?）：验证凭证 → 获得 Identity Token（临时, 5min）          │
│  • 阶段2 — 授权（Which tenant?）：选择租户 → 获得 Access Token（完整, 含 tid）         │
│  • 如果只有 1 个租户 → 自动合并为 1 步                                                  │
│                                                                                          │
│  原则3: Token 绑定租户，切换即换 Token                                                   │
│  ════════════════════════════════════════════════════════════════════════════════════   │
│  • Access Token 内含 tid（当前租户ID），所有 API 调用以此为准                            │
│  • 切换租户 = 用 Refresh Token + 新 tenantId 换发新 Access Token                        │
│  • 旧租户的 Access Token 不会被主动失效（自然过期即可）                                  │
│                                                                                          │
│  原则4: 移动端与 Web 端使用完全相同的鉴权协议                                            │
│  ════════════════════════════════════════════════════════════════════════════════════   │
│  • 移动端和 Web 端走相同的 REST API                                                     │
│  • 区别仅在 Token 存储位置：Web → HttpOnly Cookie / Mobile → Secure Storage             │
│  • 区别仅在租户解析策略：Web 可用子域名 / Mobile 只能用 JWT                             │
│                                                                                          │
└─────────────────────────────────────────────────────────────────────────────────────────┘
```

> **产品约束 — 患者/C端品牌 App 的租户选择规则**：
>
> 患者品牌 App（如 A医疗App）**默认直接进入当前服务租户**，不弹出租户选择页。
> 仅在以下条件**同时满足**时才展示选择器：
> 1. 该 Identity 在 ≥2 个租户有有效 Principal 记录
> 2. 无法通过品牌 App 或设备记忆自动判定目标租户
>
> | 入口类型 | 行为 | 是否弹选择器 |
> |---------|------|:----------:|
> | 品牌 App（如 A医疗App） | 子域名/品牌已锁定租户 → 直进 | ❌ |
> | 通用平台 App | 查 principal_list；仅 1 条 → 直进 | ❌ |
> | 通用平台 App | 查 principal_list；≥2 条 + 有设备记忆 → 高亮上次 | ⚠️ 快捷进入 |
> | 通用平台 App | 查 principal_list；≥2 条 + 无记忆 → 必须选择 | ✅ |
>
> 管理端（B端）不受此约束，始终走标准租户选择器流程。

#### 场景B：同一职员在两家租户兼职（B端 — Web+Mobile）

```text

│     • 从 A医疗(MEMBER) 切换到 C诊所(ADMIN) → 权限提升                                   │

│                                                                                          │
│  5. 移动端处理 — 应用内切换                                                              │
│     • 不需要跳转不同域名                                                                 │
│     • 在 App 内展示租户列表，选择后切换 Token                                            │
│     • 前端状态完全重置（清除缓存数据、重新加载插件数据）                                  │
│                                                                                          │
└─────────────────────────────────────────────────────────────────────────────────────────┘
```

### 3.3 完整鉴权流程（统一协议）

```text
┌─────────────────────────────────────────────────────────────────────────────────────────┐
│               统一鉴权流程（Web + Mobile 共用）                                          │
├─────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                          │
│  ┌─────────────────────────────────────────────────────────────────────────────────┐    │
│  │                    Phase 1: 身份认证（Authentication）                           │    │
│  │                                                                                  │    │
│  │  Client ──▶ POST /api/auth/login                                                │    │
│  │              { phone + smsCode }  OR  { email + password }  OR  { oauth_code }  │    │
│  │                                                                                  │    │
│  │  Server:                                                                         │    │
│  │   1. 验证凭证 → 定位 Identity                                                   │    │
│  │   2. 查询 identity 的所有活跃 tenant_member 关系 → actor_list                   │    │
│  │   3. 查询 identity 的所有活跃 tenant_principal 关系 → subject_list              │    │
│  │   4. 合并: available = actor_list ∪ subject_list（标注 role=actor/subject）      │    │
│  │   5. 分支判断:                                                                   │    │
│  │      ┌─────────────────────┐   ┌─────────────────────────────────────────────┐  │    │
│  │      │ 总数 == 0           │   │ 总数 == 1                                   │  │    │
│  │      │ → 返回 NO_TENANT    │   │ → 自动进入唯一租户                          │  │    │
│  │      │ → 引导创建/加入租户  │   │ → 根据来源表签发对应 Token（Actor/Subject） │  │    │
│  │      └─────────────────────┘   └─────────────────────────────────────────────┘  │    │
```

### 4.4 useTenant() / TenantCapability v1.2 契约定义

> **目的**：明确 useTenant() 和 TenantCapability 在引入 Actor/Subject 双轨后的最终返回结构，防止前后端字段漂移。
>
> **核心原则**：useTenant() 是租户级 Hook，不拆分 Actor/Subject；通过 `currentIdentity` 字段统一暴露当前身份信息。

```typescript
/**
 * useTenant() v1.2 — 统一租户 Hook
 * 位于: runtime-sdk-react
 *
 * ⚠️ 设计约束:
 * - 不新建 usePrincipal() — 身份信息统一由 useTenant() 暴露
 * - principalId 通过 currentIdentity.refId 获取，不作为顶层字段
 * - availableTenants 中每条记录自带 role/roleType，前端按需分组
 */

interface UseTenantResult {
  // --- 租户信息（不变） ---
  tenantId: string;
  tenant: TenantInfo;                  // id, code, name, logoUrl, status, plan

  // --- 当前身份（v1.2 新增） ---
  currentIdentity: {
    role: 'actor' | 'subject';         // 当前 Token 的角色类型
    roleType: string;                  // Actor: "OWNER"|"ADMIN"|"MEMBER"; Subject: "CUSTOMER"|"GUEST"
    refId: string;                     // Actor: memberId (mid); Subject: principalId (pid)

  switchTenant: (tenantId: string, role?: 'actor' | 'subject') => Promise<void>;
  isFeatureEnabled: (feature: string) => boolean;
}
```

```java
/**
 * TenantCapability v1.2 — 后端能力契约
```

## 5. 租户解析器补齐

### 5.1 SubdomainTenantResolver

```java
/**
 * 子域名租户解析器 — 从 Host header 提取 tenant_code
 * 
 * 模式: {tenant_code}.console.brix.com
 * 场景: Web 管理端，登录前的租户识别
 * 优先级: 200（在 JWT=100 和 Header=150 之后）
 */
@Component
@Order(200)
@ConditionalOnProperty(name = "brix.tenant.resolver.subdomain.enabled", havingValue = "true")
public class SubdomainTenantResolver implements TenantResolver {

    @Value("${brix.tenant.resolver.subdomain.pattern:^([a-z0-9-]+)\\.console\\.}")
    private String subdomainPattern;
    
    private Pattern compiledPattern;
    
    @PostConstruct
    void init() {
        compiledPattern = Pattern.compile(subdomainPattern);
    }

    @Override
    public Optional<String> resolve(HttpServletRequest request) {
        String host = request.getServerName();
        if (host == null) return Optional.empty();
        
        Matcher matcher = compiledPattern.matcher(host);
        if (matcher.find()) {
            String tenantCode = matcher.group(1);
            // 安全校验: tenant code 只允许字母数字和连字符
            if (tenantCode.matches("^[a-z0-9][a-z0-9-]{0,48}[a-z0-9]$")) {
                return tenantRepository.findTenantIdByCode(tenantCode);
            }
        }
        return Optional.empty();
    }

    @Override
    public int getOrder() { return 200; }
}
```

### 5.2 PathTenantResolver

```java
/**
 * 路径参数租户解析器 — 从 URL Path 提取 tenant_code
 * 
 * 模式: /t/{tenant_code}/...
 * 场景: 小程序、不支持子域名的场景
 * 优先级: 250
 */
@Component
@Order(250)
@ConditionalOnProperty(name = "brix.tenant.resolver.path.enabled", havingValue = "true")
public class PathTenantResolver implements TenantResolver {

    private static final Pattern PATH_PATTERN = Pattern.compile("^/t/([a-z0-9-]+)(/.*)?$");

    @Override
    public Optional<String> resolve(HttpServletRequest request) {
        String path = request.getRequestURI();
        Matcher matcher = PATH_PATTERN.matcher(path);
        if (matcher.matches()) {
            String tenantCode = matcher.group(1);
            return tenantRepository.findTenantIdByCode(tenantCode);
        }
        return Optional.empty();
    }

    @Override
    public int getOrder() { return 250; }
}
```

### 5.3 QueryTenantResolver — 本版不纳入通用解析链

> **决策**：公开注册（`/api/public/tenant/register`）、邀请链接（`/invite?tenantCode=xxx`）等场景中的 `tenantCode` 参数，
> 由对应 Controller 显式解析，**不引入通用 QueryTenantResolver**。
>
> **理由**：
> 1. Query 参数可被任意篡改，放入通用解析链会增加攻击面（所有 API 都能被注入 tenant）。
> 2. 当前需要 query 解析的场景仅限公开接口，且每个接口对 tenantCode 的校验逻辑不同（注册需验证租户可注册、邀请需验证邀请码有效性）。
> 3. 如未来确有通用 query 解析需求（如小程序 WebView 场景），可新增 `QueryTenantResolver`（`@Order(300)`），但必须配合白名单路径限制生效范围。
>
> **当前解析链优先级**：
> | 优先级 | 解析器 | 来源 | 适用场景 |
> |:------:|--------|------|----------|
> | 100 | JwtTenantResolver | JWT `tid` claim | 已认证请求（主要路径） |
> | 150 | HeaderTenantResolver | `X-Tenant-Id` Header | 开发调试 / 服务间调用 |
> | 200 | SubdomainTenantResolver | Host header | Web 管理端登录前 |
> | 250 | PathTenantResolver | URL `/t/{code}/...` | 小程序、无子域名场景 |
> | — | （不纳入链）Controller 显式处理 | Query `?tenantCode=` | 公开注册、邀请链接 |

## 10. B2B2C Actor/Subject 分离模型（路线 B）

> **来源**: B2B2C多租户模型-讨论定稿.md（路线 B — Actor/Subject 分离表）  
> **核心结论**: 新增 `sys_tenant_principal` 表存储 C 端（Subject）准入凭证，与 `sys_tenant_member`（B 端 Actor）物理分离。

> **术语说明 — 为什么叫 Principal？**
>
> `sys_tenant_principal` 是 Subject 在租户内的 **「访问主体 / 准入凭证」** 技术命名，代表的是 **「某人在某租户有被服务的资格」** 这一事实，而非业务对象本身。
>
> | 概念 | 含义 | 举例 |
> |------|------|------|
> | **Principal** | 准入凭证 — 「你是这个租户的客户」 | `sys_tenant_principal` 中一条记录 |
> | **Profile** | 租户内档案 — 「你在这个租户叫什么、偏好是什么」 | `biz_user_profile` 中一条记录 |
> | **BusinessObject** | 业务对象 — 「你为什么来、做了什么业务」 | `medical_case` / `order` / `enrollment`（插件自建表） |
>
> 三者关系：一个 Principal 对应一个 Profile，可关联 0~N 个 BusinessObject。
> Principal 的生命周期独立于 BusinessObject（见 10.9 节硬约束）。

### 10.1 角色分类

```
Platform
  ├── PlatformAdmin       — 平台运维（sys_platform_admin）
  │
  └── Tenant
       ├── Actor          — 租户运营方（B端）→ sys_tenant_member
       │   ├── OWNER      — 租户创建者 / 所有者
       │   ├── ADMIN      — 管理员
       │   └── MEMBER     — 普通职员（医生、店员、教练...）
       │
       ├── Subject        — 被服务方（C端）→ sys_tenant_principal
       │   └── CUSTOMER   — 终端用户（患者、学员、买家...）
       │
       └── Guest          — 匿名/受限访问者 → sys_tenant_principal
           └── GUEST      — 未注册但被赋予有限权限的角色
```

### 10.4 sys_tenant_member 收窄（V008 Flyway）

```sql
-- Flyway: V008__narrow_sys_tenant_member_type.sql
-- 移除 CUSTOMER/GUEST，仅保留 B 端角色

-- 1. 如果存在 CUSTOMER/GUEST 数据，先迁移到 sys_tenant_principal
INSERT INTO sys_tenant_principal (id, tenant_id, identity_id, principal_type, status, display_name, joined_at, last_access_at, created_at, updated_at)
SELECT id, tenant_id, identity_id,
       CASE WHEN member_type = 'CUSTOMER' THEN 'CUSTOMER' ELSE 'GUEST' END,
       status, display_name, joined_at, last_access_at, created_at, updated_at
FROM sys_tenant_member
WHERE member_type IN ('CUSTOMER', 'GUEST')
ON CONFLICT DO NOTHING;

-- 2. 删除已迁移的记录
DELETE FROM sys_tenant_member WHERE member_type IN ('CUSTOMER', 'GUEST');

-- 3. 更新 CHECK 约束
ALTER TABLE sys_tenant_member DROP CONSTRAINT IF EXISTS chk_member_type;
ALTER TABLE sys_tenant_member ADD CONSTRAINT chk_member_type CHECK (member_type IN ('OWNER', 'ADMIN', 'MEMBER'));
```

### 10.6 JWT 认证链适配

```
Actor（B 端）签发链 — 不变:
  sys_identity (sub) + sys_tenant_member (tid, mid, mtype) + biz_user_profile (uid)
  → JWT: { sub, tid, mid, mtype=ADMIN, uid, role=actor }

Subject（C 端）签发链 — 新增:
  sys_identity (sub) + sys_tenant_principal (tid, pid, ptype) + biz_user_profile (uid)
  → JWT: { sub, tid, pid, ptype=CUSTOMER, uid, role=subject }

收请求时（统一处理）:
  JWT → TenantFilter(解析 tid)             ← 只依赖 tid，不受影响
     → TenantContext.set(tid)              ← 不受影响
     → AuthFilter:
         if claims.has("mid") → 查 sys_tenant_member 验证有效性
         if claims.has("pid") → 查 sys_tenant_principal 验证有效性
     → SecurityContext                     ← 不受影响
     → TenantInterceptor                   ← 只依赖 tenant_id，不受影响

登录时判断逻辑:
  1. 验证凭证 → 获得 identity_id
  2. 查询 sys_tenant_member WHERE identity_id = ?  → actor_list
  3. 查询 sys_tenant_principal WHERE identity_id = ? → subject_list
  4. 合并: available_tenants = actor_list ∪ subject_list（标注角色类型）
  5. 用户选择租户 → 根据来源表签发对应类型的 JWT

同租户双身份展示（如：医生在自己诊所看病）:
  ┌─────────────────────────────────┐
  │  A 诊所（员工身份 — MEMBER）     │
  │  A 诊所（客户身份 — CUSTOMER）   │
  │  B 医院（客户身份 — CUSTOMER）   │
  └─────────────────────────────────┘
  → 前端按 tenant_id 分组，每个身份单独一行，角色标签区分
```

### 10.8 Subject 准入流程

```
Step 1: Subject 注册/登录 → 获取或创建 sys_identity
Step 2: Subject 首次接触某租户（有效准入动作之一）
  a) 租户员工手动邀请（后台添加患者/学员）
  b) Subject 接受邀请 / 确认加入租户
  c) Subject 完成注册并绑定租户
  d) 业务行为触发（如提交预约、完成报名时自动创建）
  ⚠️ 硬规则: 单纯浏览品牌页、扫码预览、访问公开链接，不自动创建 principal。
Step 3: 系统自动创建:
  • sys_tenant_principal (identity_id, tenant_id, principal_type=CUSTOMER, status=ACTIVE)
  • biz_user_profile (ref_type=PRINCIPAL, ref_id=principal_id, tenant_id, nickname=...)
  • 发布领域事件: PrincipalJoinedEvent { tenantId, principalId, principalType }
Step 4: 行业插件监听事件，按需创建 BusinessObject（Case/Enrollment/Order）
```

### 10.9 Principal 生命周期硬约束

> **硬规则**：Principal 的生命周期独立于单个 BusinessObject。
> Case/Order/Enrollment 结束时，**不自动删除或失效** Principal 记录。
> Principal 仅在以下情况失效：
> - Subject 显式退出租户（主动解绑）
> - 租户侧管理员解绑/停用该 Principal
> - Principal 状态被设为 `DISABLED` 或 `REVOKED`
